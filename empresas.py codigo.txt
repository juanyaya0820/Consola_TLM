from fastapi import APIRouter, Depends, HTTPException, status, Query
from sqlalchemy.orm import Session
from sqlalchemy import func
from typing import List, Optional
from pydantic import BaseModel
import logging

from app.db.session import get_db
from app.db import models
from app.schemas.empresas import EmpresaCreate, EmpresaResponse

router = APIRouter()
logger = logging.getLogger(__name__)

class AsignacionPayload(BaseModel):
    empresa_ids: List[int]

# ------------------------------------------------------------------------------
# 1. LISTAR EMPRESAS CON FILTRO RLS (ROW-LEVEL SECURITY)
# ------------------------------------------------------------------------------
@router.get("/", response_model=List[EmpresaResponse], summary="Listar clientes asignados")
def listar_empresas(email: Optional[str] = None, rol: Optional[str] = None, db: Session = Depends(get_db)):
    """
    Retorna únicamente el portafolio de empresas autorizadas para el usuario.
    Si el rol es 'Administrador', otorga visibilidad completa.
    """
    try:
        query = db.query(models.Empresa)
        
        if rol == 'Analista' and email:
            email_limpio = email.strip().lower()
            usuario = db.query(models.Usuario).filter(func.lower(models.Usuario.email) == email_limpio).first()
            if usuario:
                query = query.join(models.usuario_empresa).filter(
                    models.usuario_empresa.c.id_usuario == usuario.id_usuario
                )
            else:
                return []
                
        return query.order_by(models.Empresa.id_empresa.asc()).all() or []
    except Exception as e:
        logger.error(f"Error consultando empresas: {str(e)}")
        raise HTTPException(status_code=500, detail=f"Error en consulta de BD: {str(e)}")

# ------------------------------------------------------------------------------
# 2. CREACIÓN DE EMPRESAS Y VINCULACIÓN AUTOMÁTICA
# ------------------------------------------------------------------------------
@router.post("/", response_model=EmpresaResponse, status_code=status.HTTP_201_CREATED, summary="Crear cliente y asignar al creador")
def crear_empresa(
    empresa_in: EmpresaCreate, 
    creador_email: Optional[str] = Query(None), 
    db: Session = Depends(get_db)
):
    """
    Inserta la empresa en PostgreSQL y asocia automáticamente al creador 
    en la tabla intermedia 'usuario_empresa'.
    """
    try:
        nit_limpio = empresa_in.nit.strip()
        existente = db.query(models.Empresa).filter(models.Empresa.nit == nit_limpio).first()
        if existente:
            raise HTTPException(status_code=400, detail=f"Ya existe un cliente registrado con el NIT {nit_limpio}.")

        erp_final = empresa_in.software_erp or empresa_in.software_destino or "SIIGO_NUBE"

        nueva_empresa = models.Empresa(
            nombre_comercial=empresa_in.nombre_comercial.strip(),
            nit=nit_limpio,
            software_erp=erp_final
        )
        
        db.add(nueva_empresa)
        db.commit()
        db.refresh(nueva_empresa)
        
        # Auto-asignación de la empresa creada al usuario en sesión
        if creador_email:
            email_limpio = creador_email.strip().lower()
            usuario = db.query(models.Usuario).filter(func.lower(models.Usuario.email) == email_limpio).first()
            if usuario:
                usuario.empresas_asignadas.append(nueva_empresa)
                db.commit()
                logger.info(f"🔗 Empresa '{nueva_empresa.nombre_comercial}' asignada automáticamente a: {usuario.email}")

        return nueva_empresa
    except HTTPException as he:
        raise he
    except Exception as e:
        db.rollback()
        logger.error(f"Error insertando empresa: {str(e)}")
        raise HTTPException(status_code=500, detail=f"Error de persistencia: {str(e)}")

# ------------------------------------------------------------------------------
# 3. MÓDULO DE PERMISOS Y MATRIZ DE ACCESOS
# ------------------------------------------------------------------------------
@router.get("/{id_usuario}/accesos", summary="Obtener IDs de empresas asignadas")
def ver_accesos_usuario(id_usuario: int, db: Session = Depends(get_db)):
    filas = db.query(models.usuario_empresa).filter(models.usuario_empresa.c.id_usuario == id_usuario).all()
    return [fila.id_empresa for fila in filas]

@router.post("/{id_usuario}/asignar", summary="Actualizar asignación de clientes")
def asignar_empresas(id_usuario: int, payload: AsignacionPayload, db: Session = Depends(get_db)):
    try:
        db.query(models.usuario_empresa).filter(models.usuario_empresa.c.id_usuario == id_usuario).delete(synchronize_session=False)
        
        usuario = db.query(models.Usuario).filter(models.Usuario.id_usuario == id_usuario).first()
        if not usuario:
            raise HTTPException(status_code=404, detail="Usuario no encontrado.")
            
        empresas = db.query(models.Empresa).filter(models.Empresa.id_empresa.in_(payload.empresa_ids)).all()
        for emp in empresas:
            usuario.empresas_asignadas.append(emp)
            
        db.commit()
        return {"status": "success", "message": f"Se asignaron {len(empresas)} empresas al usuario."}
    except Exception as e:
        db.rollback()
        raise HTTPException(status_code=500, detail=f"Fallo al asignar permisos: {str(e)}")
